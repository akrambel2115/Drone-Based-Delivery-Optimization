\begin{frame}{Introduction \& Problem Context}
    \textbf{Problem Description:}
    A logistics company operates a fleet of drones to deliver packages from a central depot to distributed customers.
    
    \vspace{0.2cm}
    \textbf{Objective:}
    Design delivery routes for the drones such that all customers are served while \textbf{minimizing the total energy consumption}.

    \vspace{0.2cm}
    \textbf{Core Constraints:}
    \begin{itemize}
        \item Each customer must be served \textbf{exactly once}.
        \item Each drone must start and end its route at the central depot.
        \item Strict limits on drone \textbf{battery capacity} (max energy) and \textbf{payload capacity} (max weight).
        \item Routes must follow a continuous, feasible physical path.
        \item Drones must strictly avoid forbidden No-Fly Zones.
    \end{itemize}
\end{frame}

% ----------------------------------------------------------------
% 3. Modeling Assumptions
% ----------------------------------------------------------------
\begin{frame}{Modeling Assumptions}
    Beyond the core project statement constraints, we established several critical assumptions to frame the mathematical model:

    \vspace{0.3cm}
    \begin{itemize}
        \item \textbf{Offline \& Static Routing:} All customer locations and demands are known entirely in advance at $T=0$. No dynamic requests are handled mid-flight.
        \item \textbf{Homogeneous Fleet:} All deployed drones possess identical battery and payload capacities.
        \item \textbf{No Split Deliveries:} A customer's entire demand must be satisfied by a single drone visit.
        \item \textbf{Infinite Depot Supply:} The central depot has unrestricted inventory; routing is the sole bottleneck.
        \item \textbf{Anisotropic Flight Cost:} Vertical movement incurs an anisotropic energy penalty (ascending is strictly more expensive than descending).
    \end{itemize}
\end{frame}

% 4. Preprocessing: Graph Abstraction
% ----------------------------------------------------------------
\begin{frame}{Preprocessing: Graph Abstraction}
    \textbf{Challenge:} Embedding full 3D obstacle avoidance within routing formulations is computationally intractable.
    
    \vspace{0.3cm}
    \textbf{Solution: A* Pathfinding Abstraction}
    \begin{itemize}
        \item We invoke an \textbf{A* search algorithm} utilizing \textbf{26-way 3D connectivity} to compute smooth Euclidean flight paths.
        \item Computes the minimum-energy feasible trajectory between every pair of locations $(i, j)$ avoiding NFZs.
        \item Yields an energy matrix $E_{ij}$ representing the exact flight cost.
    \end{itemize}
    
    \vspace{0.3cm}
    \textbf{Result:} The physical problem reduces to routing on a complete directed graph $G = (V, A)$ where arcs $A$ are weighted by minimum energy $E_{ij}$.
\end{frame}

% ----------------------------------------------------------------
% 5. Formulation 1: Vehicle-Indexed MILP
% ----------------------------------------------------------------
\begin{frame}{Formulation 1: Vehicle-Indexed MILP}
    \textbf{Approach:} Classical Capacitated Vehicle Routing Problem with Energy Constraints.
    
    \begin{itemize}
        \item \textbf{Variables:} $x_{ijk} \in \{0, 1\}$ tracks if drone $k$ traverses arc $(i,j)$.
        \item \textbf{Miller-Tucker-Zemlin (MTZ) Constraints:}
        \begin{equation*}
            u_{ik} + d_j \le u_{jk} + M(1 - x_{ijk})
        \end{equation*}
        Simultaneously tracks cumulative delivered payload ($u_{ik}$) and eliminates disconnected subtours via strictly increasing sequences.
        \vspace{0.3cm}
        \item \textbf{Major Drawback - Vehicle Symmetry:} 
        Because the fleet is homogeneous, any permutation of drone indices yields the exact same objective value, causing severe branch-and-bound tree explosion.
    \end{itemize}
\end{frame}

% ----------------------------------------------------------------
% 6. Formulation 2: Two-Commodity Network Flow
% ----------------------------------------------------------------
\begin{frame}{Formulation 2: Two-Commodity Network Flow}
    \textbf{Approach:} A \emph{vehicle-agnostic} perspective tracking two continuous "fluids".
    
    \begin{itemize}
        \item \textbf{Variables:} $x_{ij} \in \{0, 1\}$ (drone-independent assignment).
        \item \textbf{Commodity 1 ($F_{ij}$):} Payload flow. Balance constraint:
        \begin{equation*}
            \sum_{i} F_{ij} - \sum_{i} F_{ji} = d_j, \quad \forall\, j \in C
        \end{equation*}
        \item \textbf{Commodity 2 ($Q_{ij}$):} Energy flow. Accumulates energy cost $E_{ij}$ at each node to enforce battery capacities ($Q_{ij} \le B_{\max}$).
    \end{itemize}

    \vspace{0.1cm}
    \textbf{Key Strengths:} Eliminates drone-index symmetry completely and provides a much \textbf{tighter LP Relaxation} for exact solvers compared to MTZ.
\end{frame}

% ----------------------------------------------------------------
% 7. Computational Complexity
% ----------------------------------------------------------------
\begin{frame}{Computational Complexity}
    \textbf{Theorem:} The Drone-Based Delivery Optimization Problem is strongly NP-Hard.
    
    \vspace{0.5cm}
    \textbf{Proof Outline (By Restriction):}
    \begin{itemize}
        \item Consider a specialized instance of our problem where:
        \begin{enumerate}
            \item All $Z$-coordinates (altitudes) are identically 0.
            \item The set of No-Fly Zones is empty.
            \item The drone battery capacity is set to infinity ($B_{\max} \to \infty$).
        \end{enumerate}
        \item The anisotropic energy model reduces to isotropic 2D Euclidean distance.
        \item The problem identically reduces to the classical \textbf{Capacitated Vehicle Routing Problem (CVRP)}, which is proven to be strongly NP-Hard.
    \end{itemize}
\end{frame}

% ----------------------------------------------------------------
% 8. Exact Method: Branch and Bound (Overview)
% ----------------------------------------------------------------
\begin{frame}{Exact Method: Branch and Bound (1/2)}

To mathematically prove optimal routes, we implemented a custom exact solver based on the Two-Commodity Network Flow formulation.

\vspace{0.4cm}

\textbf{Algorithm Mechanics:}
\begin{itemize}
    \item \textbf{Initialization:} Uses a greedy \emph{Nearest Neighbor} heuristic to warm-start the search and establish a strong initial \textbf{Upper Bound (UB)}.

    \item \textbf{Search Strategy:} Depth-First Search (DFS) over the binary assignment variables $x_{ij}$.

    \item \textbf{Bounding Oracle:} At every node in the search tree, we solve the \textbf{Linear Programming (LP) Relaxation} using the PuLP library to compute a strong \textbf{Lower Bound (LB)}.
\end{itemize}

\end{frame}

% 10. Metaheuristics (SA & GA)
% ----------------------------------------------------------------
\begin{frame}{Metaheuristic 1: Simulated Annealing (SA)}
    Because B\&B scales exponentially, we developed robust metaheuristics to find high-quality solutions rapidly for larger instances.
    
    \vspace{0.3cm}
    \textbf{Simulated Annealing Mechanics:}
    \begin{itemize}
        \item \textbf{Neighborhood Operators:} Uses both intra-route and inter-route perturbations (Swap, Relocate, Inversion) to explore adjacent solutions.
        \item \textbf{Cooling Schedule:} Employs a geometric temperature decay to probabilistically accept worse solutions early on, effectively escaping local minima.
        \item \textbf{Exploitation:} As temperature drops, the algorithm converges into a strict hill-climbing descent.
    \end{itemize}
\end{frame}

% ----------------------------------------------------------------
% 11. Metaheuristic 2: Genetic Algorithm (GA)
% ----------------------------------------------------------------
\begin{frame}{Metaheuristic 2: Genetic Algorithm (GA)}
    \textbf{Genetic Algorithm Mechanics:}
    \begin{itemize}
        \item \textbf{Strong Initialization:} We seed the initial population with high-quality constructive heuristics (\textbf{Clarke \& Wright Savings} and \textbf{Nearest Neighbour}), enabling extremely rapid convergence.
        \item \textbf{Population Management:} Maintains a diverse pool of chromosomes representing "giant tours" (customer permutations).
        \item \textbf{Crossover:} Uses Order Crossover (OX) to preserve optimal relative customer sequences from parent routes.
        \item \textbf{Split Decoder:} Translates the sequence into optimal drone routes respecting payload and battery capacities.
    \end{itemize}
\end{frame}
